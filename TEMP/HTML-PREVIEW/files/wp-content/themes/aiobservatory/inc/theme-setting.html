function get_aio_page_title(){
    $html = '';
    $count = 0;
     $title = explode(" ", get_the_title());
     foreach ($title as $value) {
        if($count==1){
        $html.=' <span>'.$value.'</span> ';
    } else {
        $html.=' '.$value.' ';
    }

        $count++;
     }
     return $html;
 }

function get_aio_style_title($text){
    $html = '';
